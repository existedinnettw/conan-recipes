cmake_minimum_required(VERSION 3.15)
project(test_package LANGUAGES CXX)

option(O3DE_TEST_ATOM "The package has the Atom Gems" OFF)
option(O3DE_TEST_GEM_MODULES "Gem modules are linked (monolithic build)" OFF)
option(O3DE_TEST_EDITOR "The package has the editor libraries" OFF)

find_package(o3de REQUIRED CONFIG)

add_executable(test_package test_package.cpp)
target_link_libraries(test_package PRIVATE AZ::AzFramework)
target_compile_features(test_package PRIVATE cxx_std_20)
if(O3DE_TEST_ATOM)
  target_compile_definitions(test_package PRIVATE O3DE_TEST_ATOM)
  target_link_libraries(test_package PRIVATE Gem::Atom_RHI.Reflect)
  if(O3DE_TEST_GEM_MODULES)
    target_compile_definitions(test_package PRIVATE O3DE_TEST_GEM_MODULES)
    target_link_libraries(test_package PRIVATE Gem::Atom_RPI.Private)
  endif()
endif()
if(O3DE_TEST_EDITOR)
  target_compile_definitions(test_package PRIVATE O3DE_TEST_EDITOR)
  target_link_libraries(test_package PRIVATE AZ::AzQtComponents AZ::AzToolsFramework)
endif()
